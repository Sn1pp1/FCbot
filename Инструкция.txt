📦 ДЕПЛОЙ
================================================================================

# --- 1. Подготовка и клонирование ---
sudo apt update
sudo apt install python3-pip python3-venv git -y
cd /opt
git clone https://github.com/Sn1pp1/FCbot.git
cd FCbot

# --- 2. Виртуальное окружение и зависимости ---
python3 -m venv venv
source venv/bin/activate
pip install --upgrade pip
pip install -r requirements.txt python-dotenv

# --- 3. Настройка токена (через nano) ---
nano .env

# Вставьте одну строку в формате:
BOT_TOKEN=ваш_токен_от_botfather

# Пример формата (не копируйте, это просто пример!):
# BOT_TOKEN=123456789:AAHabcdefghijklmnopqrstuvwxyz123

# Сохраните: Ctrl+O → Enter → Выход: Ctrl+X

# Проверьте:
cat .env

# --- 4. Тестовый запуск (опционально) ---
python main.py
# Увидели "Start polling"? Нажмите Ctrl+C

# --- 5. Создание systemd сервиса ---
sudo tee /etc/systemd/system/fcbot.service > /dev/null << 'EOF'
[Unit]
Description=FCbot Telegram Bot
After=network.target
StartLimitBurst=3
StartLimitIntervalSec=60

[Service]
Type=simple
User=root
WorkingDirectory=/opt/FCbot
Environment="PATH=/opt/FCbot/venv/bin"
Environment="PYTHONUNBUFFERED=1"
ExecStart=/opt/FCbot/venv/bin/python /opt/FCbot/main.py
Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target
EOF

# --- 6. Активация и запуск ---
sudo systemctl daemon-reload
sudo systemctl enable fcbot
sudo systemctl start fcbot
sudo systemctl status fcbot
# ✅ Должно быть: Active: active (running)
